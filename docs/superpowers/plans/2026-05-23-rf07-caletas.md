'use server'

import { createClient } from '@/lib/supabase/server'
import { prisma } from '@/lib/prisma'
import type { ActionResult } from '@/types'
import type { CaletaModel } from '@/generated/prisma'

type CreateCaletaInput = {
  nombre: string
  icono: string        // emoji string
  color: string        // hex color
  montoObjetivo: number // pesos COP — SA convierte ×100
  fechaObjetivo?: string // YYYY-MM-DD, opcional
  descripcion?: string
}

export async function createCaleta(
  input: CreateCaletaInput
): Promise<ActionResult<CaletaModel>> {
  const supabase = await createClient()
  const {
    data: { user },
  } = await supabase.auth.getUser()
  if (!user) return { success: false, error: 'No autorizado' }

  const prismaUser = await prisma.user.findUnique({ where: { supabaseId: user.id } })
  if (!prismaUser) return { success: false, error: 'Usuario no encontrado' }

  if (!input.nombre.trim()) return { success: false, error: 'El nombre es obligatorio' }
  if (input.montoObjetivo <= 0) return { success: false, error: 'El objetivo debe ser mayor a 0' }

  try {
    const caleta = await prisma.caleta.create({
      data: {
        userId: prismaUser.id,
        nombre: input.nombre.trim(),
        icono: input.icono,
        color: input.color,
        montoObjetivo: Math.round(input.montoObjetivo * 100),
        fechaObjetivo: input.fechaObjetivo ? new Date(input.fechaObjetivo) : null,
        descripcion: input.descripcion?.trim() || null,
      },
    })
    return { success: true, data: caleta }
  } catch {
    return { success: false, error: 'Error al crear la caleta' }
  }
}

'use server'

import { createClient } from '@/lib/supabase/server'
import { prisma } from '@/lib/prisma'
import type { ActionResult } from '@/types'
import type { CaletaDepositModel } from '@/generated/prisma'

export async function createCaletaDeposit(
  caletaId: string,
  montoPesos: number,
  nota?: string
): Promise<ActionResult<CaletaDepositModel>> {
  const supabase = await createClient()
  const {
    data: { user },
  } = await supabase.auth.getUser()
  if (!user) return { success: false, error: 'No autorizado' }

  const prismaUser = await prisma.user.findUnique({ where: { supabaseId: user.id } })
  if (!prismaUser) return { success: false, error: 'Usuario no encontrado' }

  if (montoPesos <= 0) return { success: false, error: 'El aporte debe ser mayor a 0' }

  const caleta = await prisma.caleta.findFirst({
    where: { id: caletaId, userId: prismaUser.id },
  })
  if (!caleta) return { success: false, error: 'Caleta no encontrada' }

  const montoCentavos = Math.round(montoPesos * 100)

  try {
    const deposit = await prisma.$transaction(async (tx) => {
      const d = await tx.caletaDeposit.create({
        data: {
          caletaId,
          monto: montoCentavos,
          nota: nota?.trim() || null,
        },
      })

      const nuevoMonto = caleta.montoActual + montoCentavos
      await tx.caleta.update({
        where: { id: caletaId },
        data: {
          montoActual: nuevoMonto,
          completada: nuevoMonto >= caleta.montoObjetivo,
        },
      })

      return d
    })

    return { success: true, data: deposit }
  } catch {
    return { success: false, error: 'Error al registrar el aporte' }
  }
}

import { prisma } from '@/lib/prisma'

  const supabase = await createClient()
  const {
    data: { user },
  } = await supabase.auth.getUser()

import type { CaletaModel, CaletaDepositModel } from '@/generated/prisma'
import { ChevronRight } from 'lucide-react'

export type CaletaWithDeposits = CaletaModel & {
  deposits: CaletaDepositModel[]
}

const fmt = (centavos: number) =>
  new Intl.NumberFormat('es-CO', {
    style: 'currency',
    currency: 'COP',
    maximumFractionDigits: 0,
  }).format(centavos / 100)

type Props = {
  caleta: CaletaWithDeposits
  onTap: (caleta: CaletaWithDeposits) => void
}

export function CaletaCard({ caleta, onTap }: Props) {
  const pct =
    caleta.montoObjetivo > 0
      ? Math.min(100, Math.round((caleta.montoActual / caleta.montoObjetivo) * 100))
      : 0
  const sinAportes = caleta.montoActual === 0

  return (
    <button
      type="button"
      onClick={() => onTap(caleta)}
      className="w-full rounded-2xl border border-border bg-card p-4 text-left transition-opacity active:opacity-80"
    >
      {/* Fila superior */}
      <div className="flex items-center gap-3 mb-3">
        <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-muted text-2xl">
          {caleta.icono}
        </div>
        <div className="flex-1 min-w-0">
          <p className="text-sm font-semibold text-foreground truncate">{caleta.nombre}</p>
          <p className="text-xs text-muted-foreground mt-0.5">
            {caleta.completada
              ? '¡Meta cumplida! 🎉'
              : sinAportes
              ? 'Sin aportes aún'
              : `${pct}% completo`}
          </p>
        </div>
        <ChevronRight className="h-4 w-4 shrink-0 text-muted-foreground" />
      </div>

      {/* Barra de progreso */}
      <div className="h-2 rounded-full bg-muted overflow-hidden mb-3">
        <div
          className="h-full rounded-full transition-all"
          style={{ width: `${pct}%`, backgroundColor: caleta.color }}
        />
      </div>

      {/* Montos */}
      <div className="flex items-center justify-between">
        <span className="text-sm font-semibold tabular-nums text-foreground">
          {fmt(caleta.montoActual)}
        </span>
        <span className="text-xs tabular-nums text-muted-foreground">
          de {fmt(caleta.montoObjetivo)}
        </span>
      </div>
    </button>
  )
}

        {mode === 'create' && (
          <CreateForm onClose={handleClose} onSuccess={onSuccess} />
        )}
        {mode === 'detail' && caleta && !depositMode && (
          <DetailView
            caleta={caleta}
            onDeposit={() => setDepositMode(true)}
            onDelete={onSuccess}
            onClose={handleClose}
          />
        )}
        {mode === 'detail' && caleta && depositMode && (
          <DepositForm
            caleta={caleta}
            onBack={() => setDepositMode(false)}
            onSuccess={onSuccess}
            onClose={handleClose}
          />
        )}
      </DrawerContent>
    </Drawer>
  )
}

'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import { CaletaCard } from './caleta-card'
import { CaletaDrawer } from './caleta-drawer'
import type { CaletaWithDeposits } from './caleta-card'

const fmt = (centavos: number) =>
  new Intl.NumberFormat('es-CO', {
    style: 'currency',
    currency: 'COP',
    maximumFractionDigits: 0,
  }).format(centavos / 100)

type Props = {
  caletas: CaletaWithDeposits[]
}

export function CaletasClient({ caletas }: Props) {
  const router = useRouter()
  const [drawerOpen, setDrawerOpen] = useState(false)
  const [drawerMode, setDrawerMode] = useState<'create' | 'detail'>('create')
  const [selectedCaleta, setSelectedCaleta] = useState<CaletaWithDeposits | undefined>()

  const totalAhorrado = caletas.reduce((s, c) => s + c.montoActual, 0)
  const totalObjetivo = caletas.reduce((s, c) => s + c.montoObjetivo, 0)

    setSelectedCaleta(undefined)
    setDrawerOpen(true)

    setSelectedCaleta(caleta)
    setDrawerOpen(true)

        <button
          type="button"
          onClick={openCreate}
          className="rounded-full bg-primary px-4 py-1.5 text-sm font-semibold text-primary-foreground"
        >
          + Nueva
        </button>
      </div>

      {/* Summary card */}
      {caletas.length > 0 && (
        <div className="rounded-2xl border border-border bg-card p-4 mb-5">
          <p className="text-xs text-muted-foreground mb-1">Ahorrado en caletas</p>
          <p className="text-2xl font-bold tabular-nums text-foreground">{fmt(totalAhorrado)}</p>
          <p className="text-xs text-muted-foreground mt-1">
            de <span className="tabular-nums">{fmt(totalObjetivo)}</span> meta total
          </p>
        </div>
      )}

      {/* Empty state */}
      {caletas.length === 0 && (
        <div className="flex flex-col items-center justify-center py-20 text-center gap-3">
          <span className="text-5xl">🐷</span>
          <p className="text-sm text-muted-foreground">Aún no tienes caletas.</p>
          <button
            type="button"
            onClick={openCreate}
            className="rounded-full bg-primary px-5 py-2 text-sm font-semibold text-primary-foreground"
          >
            Crear mi primera caleta
          </button>
        </div>
      )}

        ))}

      <CaletaDrawer

        mode={drawerMode}
        caleta={selectedCaleta}
        onClose={() => setDrawerOpen(false)}
        onSuccess={handleSuccess}
      />
    </div>
  )
}

import { redirect } from 'next/navigation'
import { createClient } from '@/lib/supabase/server'
import { prisma } from '@/lib/prisma'
import { CaletasClient } from '@/components/features/caletas/caletas-client'

export default async function CaletasPage() {
  const supabase = await createClient()
  const {
    data: { user: supabaseUser },
  } = await supabase.auth.getUser()
  if (!supabaseUser) redirect('/login')

  const prismaUser = await prisma.user.findUnique({
    where: { supabaseId: supabaseUser.id },
  })
  if (!prismaUser) redirect('/login')

  const caletas = await prisma.caleta.findMany({
    where: { userId: prismaUser.id },
    include: {
      deposits: {
        orderBy: { createdAt: 'desc' },
        take: 5,
      },
    },
    orderBy: { createdAt: 'desc' },
  })

  return (
    <main className="pb-24 px-4 pt-6">
      <CaletasClient caletas={caletas} />
    </main>
  )
}
